;; The C33 PE has a 32-bit instruction address space.  A branch or call with
;; one ext prefix carries a signed 22-bit *byte* displacement (including its
;; fixed low zero bit), so it reaches -2 MiB through 2 MiB - 2.  Two ext
;; prefixes carry the full signed 32-bit displacement.

;; Condition codes
;;
;; Data movement (load, store, register copy) does not modify condition
;; codes.  But there is no way to add two registers together without
;; modifying the condition codes.
;;
;; So we do not expose the condition codes until after reload.  The
;; general approach is to have a define_insn_and_split for the basic
;; operation with no condition codes in the pattern (to give the
;; optimizers maximal freedom).  The splitter adds a clobber of the
;; condition codes after reload.  There is a distinct pattern which
;; sets the condition codes.
;;
;; As noted, data movement does not affect condition codes.
;;
;; Arithmetic generally set the codes in the expected ways, with mul
;; instructions being a notable outlier.  div instructions generally
;; do the right thing, except when the output registers are the same
;; when the flags do not get set.  We just assume they're clobbered
;; for div instructions to avoid MD bloat with marginal benefit
;;
;; The bit manipulation instructions (clr1, not1, set1) set condition
;; codes, but not in a useful way (they're set to the prior status of
;; the bit).  So we just model those as clobbers.  tst1 does set the
;; condition codes in a useful way.  We could perhaps do better with
;; these by noting they only modify the Z flag, it doesn't seem worth
;; the effort.
;;
;; Byte swaps seem to change the condition codes, but I haven't tried
;; to describe how.
;;
;; I have no documentation on the rotate instructions.  They likely
;; set the condition codes, but I've left them as clobbers for now.

;; Delay slots.
;;
;; Branches, jumps and calls have one slot, selected by a ".d" suffix on the
;; mnemonic and printed by %#.  It is *not* annulled: the slot instruction
;; runs whether or not a conditional branch is taken, which is why the
;; define_delay below has (nil) (nil) for the annul-true and annul-false
;; slots.  Leaving a branch unfilled is always safe -- %# then prints
;; nothing and the plain form is emitted.
;;
;; What may go in a slot is decided by length and type.  Every two-byte
;; pattern here is a bare mnemonic, while everything wider is an x-prefixed
;; form whose ext prefixes the assembler synthesises.  The PE manual also
;; expressly forbids memory access, even for a two-byte load or store, so
;; load, store and memory-bit types are excluded below.  The remaining
;; exceptions are marked "no" one by one at their patterns:
;;
;;   ret, reti      control transfer of their own
;;   pushn, popn    touch the stack
;;   add/sub %sp    likewise
;;   ld.w %sp,%rs   likewise
;;
;; and one more that depends on *which* kind of branch is being filled, so
;; it cannot be a property of the pattern alone.  A call pushes the return
;; address before its slot runs:
;;
;;     call.d  f          ; %sp -= 4, return address stored
;;     ld.w    [%sp+0],%r4  ; ...lands on top of it
;;
;; so in a call's slot every %sp reference is off by four, and [%sp+0]
;; overwrites the address the call just pushed -- the callee returns into
;; whatever the slot happened to store.  Jumps and conditional branches do
;; not touch %sp and are unaffected.
;;
;; The call patterns do not mention %sp in their RTL, so the delay-slot
;; pass cannot see this for itself; teaching them to would be the more
;; honest fix, at the cost of perturbing every pass that looks at a call.
;; Two eligibility rules are the cheaper one.

(define_attr "needs_delay_slot" "yes,no" (const_string "no"))

;; The width test is spelled with match_test rather than the more obvious
;; (eq_attr "length" "2").  Patterns whose length is computed by a function
;; -- the moves, via c33_move_length -- make genattrtab substitute the call
;; for the attribute and then emit the comparison against the enumerator
;; name LENGTH_2, which does not exist for a numeric attribute.  Calling
;; get_attr_length outright sidesteps that and means the same thing.
(define_attr "in_delay_slot" "yes,no"
  (cond [(eq_attr "needs_delay_slot" "yes") (const_string "no")
	 (eq_attr "type" "load,store,bit1") (const_string "no")
	 (match_test "get_attr_length (insn) == 2") (const_string "yes")]
	(const_string "no")))

;; Eligible for a call's slot: everything above, less anything touching
;; %sp.  Note this is not the same as the "ld.w %sp,%rs" exclusion, which
;; is about the pattern; this is about the operands, since a stack local is
;; an ordinary two-byte ld that only becomes unsafe here.
(define_attr "in_call_delay_slot" "yes,no"
  (cond [(eq_attr "in_delay_slot" "no") (const_string "no")
	 (match_test "c33_uses_sp_p (insn)") (const_string "no")]
	(const_string "yes")))

(define_delay (and (eq_attr "needs_delay_slot" "yes")
		   (match_test "CALL_P (insn)"))
  [(eq_attr "in_call_delay_slot" "yes") (nil) (nil)])

(define_delay (and (eq_attr "needs_delay_slot" "yes")
		   (not (match_test "CALL_P (insn)")))
  [(eq_attr "in_delay_slot" "yes") (nil) (nil)])

(define_expand "movqi"
  [(set (match_operand:QI 0 "general_operand" "")
	(match_operand:QI 1 "general_operand" ""))]
  ""
  {
    /* One side has to be a register: there is no hardwired zero register, so
       a store of a constant has to load it first.  */
    if (!register_operand (operand0, QImode)
	&& !register_operand (operand1, QImode))
      operands[1] = copy_to_mode_reg (QImode, operand1);
  })

;; Reload can expose the low part of a pointer-valued %sp copy as a QImode
;; move.  %sp is a system register and has only the word-sized ld.w transfer;
;; using that transfer is correct because the destination's upper bits are
;; unspecified for a QImode value.  Match the hard register literally so the
;; generic "r" alternatives below never claim an impossible %sp operand.
(define_insn "*movqi_from_sp"
  [(set (match_operand:QI 0 "register_operand" "=r")
	(reg:QI SP_REGNUM))]
  ""
  "ld.w\t%0,%1"
  [(set_attr "length" "2")
   (set_attr "in_delay_slot" "no")
   (set_attr "type" "other")])

(define_insn "*movqi_internal"
  [(set (match_operand:QI 0 "nonimmediate_operand" "=r,r,r,Q,r,m")
	(match_operand:QI 1 "general_operand"       " r,i,Q,r,m,r"))]
  "register_operand (operands[0], QImode)
   || register_operand (operands[1], QImode)"
{
  return output_move_single (operands);
}
  [(set (attr "length") (symbol_ref "c33_move_length (insn)"))
   (set_attr "type" "other,other,load,store,load,store")])

;; movsi and helpers

(define_expand "movsi"
  [(set (match_operand:SI 0 "general_operand" "")
	(match_operand:SI 1 "general_operand" ""))]
  ""
  {
    /* One side has to be a register.  There is no hardwired zero register on
       this target, so unlike the V850 a store of 0 is not a special case --
       the constant has to be loaded first.  */
    if (!register_operand (operands[0], SImode)
	&& !register_operand (operands[1], SImode))
      operands[1] = copy_to_mode_reg (SImode, operands[1]);

    /* No HIGH/LO_SUM splitting: xld.w takes a full 32-bit immediate and the
       assembler emits the two ext prefixes it needs.  The V850 had to build
       wide constants with movhi/movea because it had no such form.  */
  })

;; %sp is system register 1, so moving it uses the ld.w special-register
;; forms (RD,SS and SD,RS2).  Same mnemonic and same syntax as a general
;; register copy -- the assembler picks the encoding from the operand -- so
;; these are just two more alternatives rather than separate patterns.
;;
;; They have to exist.  When the frame or arg pointer is eliminated to %sp
;; and the result must reach an "r" operand, this is the only way to get it
;; there; without it LRA reloads the copy, the reload needs the same copy,
;; and it recurses until it hits the reload limit.

(define_insn "*movsi_internal"
  [(set (match_operand:SI 0 "nonimmediate_operand" "=r,r,r,Q,r,m,r,f")
	(match_operand:SI 1 "movsi_source_operand"  " r,i,Q,r,m,r,f,r"))]
  "register_operand (operands[0], SImode)
   || register_operand (operands[1], SImode)"
{
  return output_move_single (operands);
}
  [(set (attr "length") (symbol_ref "c33_move_length (insn)"))
   ;; "*" means the attribute's own default, which derives eligibility
   ;; from length -- the right answer everywhere except the two %sp
   ;; alternatives, which are two bytes but must still be excluded (see
   ;; the delay-slot comment above).
   (set_attr "in_delay_slot" "*,*,*,*,*,*,no,no")
   (set_attr "type" "other,other,load,store,load,store,other,other")])

;; Floating point is soft-float only, so an SFmode value is just a 32-bit
;; pattern in a general register and moves the same way as SImode.

(define_insn "*movsf_internal"

  "register_operand (operands[0], SFmode)
   || register_operand (operands[1], SFmode)"
{
  return output_move_single (operands);
}
  [(set (attr "length") (symbol_ref "c33_move_length (insn)"))
   (set_attr "type" "other,other,load,store,load,store")])

;; Stack pointer arithmetic.
;;
;; %sp is not a general register, and the C33 has dedicated forms for moving
;; it: add/sub %sp,imm10 scale the immediate by 4 and cannot be extended with
;; ext, so one instruction reaches +/-4092 bytes (core manual p64).  Anything
;; larger has to go through a general register and ld.w %sp,%rs.

(define_insn "add_sp_imm"
  [(set (reg:SI SP_REGNUM)
	(plus:SI (reg:SI SP_REGNUM)
		 (match_operand:SI 0 "const_int_operand" "i")))]
  "IN_RANGE (INTVAL (operands[0]), -C33_MAX_SP_ADJUST, C33_MAX_SP_ADJUST)
   && (INTVAL (operands[0]) & 3) == 0"
{
  HOST_WIDE_INT n = INTVAL (operands[0]);
  operands[0] = GEN_INT ((n < 0 ? -n : n) / 4);
  return n < 0 ? "sub\t%%sp,%0" : "add\t%%sp,%0";
}
  [(set_attr "length" "2")
   (set_attr "in_delay_slot" "no")])

;; Generic code -- argument pushing, alloca, stack probes -- reaches the stack
;; pointer through gen_addsi3, which after splitting is this same set wrapped
;; in a parallel with a CC clobber.  Match it too, and first, or addsi3 claims
;; it and then fails constraint checking: register_operand accepts %sp now
;; that it belongs to a register class.
;;
;; The clobber is a harmless over-approximation.  add/sub %sp,imm10 affect no
;; flags at all (core manual p64).

(define_insn "*add_sp_imm_clobber_flags"
  [(set (reg:SI SP_REGNUM)
	(plus:SI (reg:SI SP_REGNUM)
		 (match_operand:SI 0 "const_int_operand" "i")))
   (clobber (reg:CC CC_REGNUM))]
  "IN_RANGE (INTVAL (operands[0]), -C33_MAX_SP_ADJUST, C33_MAX_SP_ADJUST)
   && (INTVAL (operands[0]) & 3) == 0"
{
  HOST_WIDE_INT n = INTVAL (operands[0]);
  operands[0] = GEN_INT ((n < 0 ? -n : n) / 4);
  return n < 0 ? "sub\t%%sp,%0" : "add\t%%sp,%0";
}
  [(set_attr "length" "2")
   (set_attr "in_delay_slot" "no")])

;; Frames deeper than 4092 bytes.  add/sub %sp,imm10 cannot be extended, so
;; the adjustment has to be done in a general register.  xadd/xsub %rd,imm32
;; take any 32-bit constant (the assembler supplies the ext prefixes), so one
;; scratch is enough -- there is no need to materialise the delta separately.
;;
;; %r14 is call-clobbered and holds neither an incoming argument nor a return
;; value, so it is free in both the prologue and the epilogue.

(define_insn "add_sp_big"
  [(set (reg:SI SP_REGNUM)
	(plus:SI (reg:SI SP_REGNUM)
		 (match_operand:SI 0 "const_int_operand" "i")))
   (clobber (reg:SI 14))]
  ""
{
  HOST_WIDE_INT n = INTVAL (operands[0]);
  operands[0] = GEN_INT (n < 0 ? -n : n);
  return n < 0
	 ? "ld.w\t%%r14,%%sp\n\txsub\t%%r14,%0\n\tld.w\t%%sp,%%r14"
	 : "ld.w\t%%r14,%%sp\n\txadd\t%%r14,%0\n\tld.w\t%%sp,%%r14";
}
  [(set_attr "length" "10")])

;; Conditional branches come in three reaches -- jr<cc> disp8 (2 bytes),
;; sjr<cc> disp22 (4) and xjr<cc> disp32 (6) -- so unlike the V850 there is
;; never any need to invert the condition and jump over an unconditional
;; jump.  The assembler emits only the ext prefixes the displacement needs,
;; so picking by range here is about getting the length attribute right.

(define_insn "*branch_normal"
  [(set (pc)
	(if_then_else (match_operator 1 "comparison_operator"
				      [(reg CC_REGNUM) (const_int 0)])
		      (label_ref (match_operand 0 "" ""))
		      (pc)))]
  "reload_completed"
{
  switch (get_attr_length (insn))
    {
    case 2:  return "jr%b1%#\t%l0";
    case 4:  return "sjr%b1%#\t%l0";
    default: return "xjr%b1%#\t%l0";
    }
}
 ;; The architectural ranges are -256..254 for jr and
 ;; -2097152..2097150 for sjr (PE manual 5.6.2).  Use a conservative upper
 ;; bound for each because a forward reference is measured from the end of
 ;; the current SEQUENCE, while the C33 displacement is measured from the
 ;; branch itself.  The largest previous-pass sequence is an xjr plus its
 ;; two-byte slot, eight bytes; backward branches lose the same margin so a
 ;; single symmetric expression remains safe.
 ;;
 ;; For a forward branch
 ;; (pc) is insn_current_reference_address, which final.cc computes as
 ;; insn_last_address + the length of the whole SEQUENCE -- the address
 ;; *after* the branch and its delay slot.  The C33 measures the
 ;; displacement from the branch itself, so the real distance is farther
 ;; than this expression sees.  Backward branches need no correction.
 ;;
 ;; Getting this wrong is not silent -- gas rejects a jr it cannot reach
 ;; with "operand out of range" -- but it had been wrong since before
 ;; delay slots were filled at all, and only became reachable once the
 ;; offset grew from two bytes to four.  The 4-byte forms are good for at
 ;; least 8 MB, so only the short one needs the margin.
 [(set (attr "length")
    (if_then_else (lt (abs (minus (match_dup 0) (pc)))
		      (const_int 252))
		  (const_int 2)
		  (if_then_else (lt (abs (minus (match_dup 0) (pc)))
		      (const_int 2097144))
		      (const_int 4)
		      (const_int 6))))
  (set_attr "needs_delay_slot" "yes")])

(define_expand "casesi"

   (match_operand:SI 1 "register_operand" "")
   (match_operand:SI 2 "register_operand" "")
   (match_operand 3 "" "") (match_operand 4 "" "")]
  ""
  {
    rtx reg = gen_reg_rtx (SImode);
    rtx tableaddress = gen_reg_rtx (SImode);
    rtx test;
    rtx mem;

    /* Subtract the lower bound from the index.  */
    emit_insn (gen_subsi3 (reg, operands[0], operands[1]));

    /* Compare the result against the number of table entries;
       branch to the default label if out of range of the table.  */
    test = gen_rtx_fmt_ee (GTU, VOIDmode, reg, operands[2]);
    emit_jump_insn (gen_cbranchsi4 (test, reg, operands[2], operands[4]));

    emit_insn (gen_ashlsi3 (reg, reg, GEN_INT (2)));
    /* Load the table address into a pseudo.  */
    emit_insn (gen_movsi (tableaddress,
			  gen_rtx_LABEL_REF (Pmode, operands[3])));
    /* Add the table address to the index.  */
    emit_insn (gen_addsi3 (reg, reg, tableaddress));

       c33.h for why it cannot be.  */
    mem = gen_const_mem (Pmode, reg);
    emit_insn (gen_movsi (reg, mem));
    /* Branch to the switch label.  */
    emit_jump_insn (gen_tablejump (reg, operands[3]));
    DONE;
  })

;; rl/rr take either a register or a 5-bit immediate, and 0..31 is every
;; rotate amount a 32-bit value has, so unlike the shifts there is no
;; extended form to fall back on -- both alternatives are a bare two-byte
;; instruction.  The N constraint keeps a wider constant out; GCC then puts
;; it in a register, which rotates by the low bits and is what C means.
;;
;; Both set N and Z (set_nz in the core), hence the clobber.  Without these
;; patterns a rotate became shift/shift/or: gcc 3.3.2 emitted rl eleven
;; times in wiki.app where this backend emitted none.

(define_insn_and_split "rotlsi3"
  [(set (match_operand:SI 0 "register_operand" "=r,r")
	(rotate:SI (match_operand:SI 1 "register_operand" "0,0")
		   (match_operand:SI 2 "nonmemory_operand" "r,N")))]
  ""
  "#"
  "reload_completed"
  [(parallel [(set (match_dup 0) (rotate:SI (match_dup 1) (match_dup 2)))
	      (clobber (reg:CC CC_REGNUM))])])

;; Block save/restore of the callee-saved registers.
;;
;; pushn %rN pushes the whole block %r0..%rN, decrementing %sp by 4 for each
;; (core manual 2.4.2); popn %rN is the inverse (2.4.3).  Operand 0 is N.
;; The effect on %sp and memory is described by the unspec_volatile plus the
;; explicit clobber, which is enough for the frame-related notes the
;; prologue attaches.

(define_insn "pushn"
  [(unspec_volatile [(match_operand 0 "const_int_operand" "i")]
		    UNSPECV_PUSHN)
   (clobber (reg:SI SP_REGNUM))]
  ""
  "pushn\t%%r%0"
  [(set_attr "length" "2")
   (set_attr "in_delay_slot" "no")])

(define_insn "popn"
  [(unspec_volatile [(match_operand 0 "const_int_operand" "i")]
		    UNSPECV_POPN)
   (clobber (reg:SI SP_REGNUM))]
  ""
  "popn\t%%r%0"
  [(set_attr "length" "2")
   (set_attr "in_delay_slot" "no")])

;; An interrupt pushes both PC and PSR, and reti pops both (core manual
;; 2.4.4).  There is nothing else to do: compute_register_save_size already

;; Everything V850 had here -- callt_save_interrupt, save_all_interrupt and

;; registers that do not exist on this target (r30 was ep; with

;; rejects).  None of it was reachable: nothing in c33.cc ever generated it.
