;; Condition codes
;;
;; Data movement (load, store, register copy) does not modify condition
;; codes.  But there is no way to add two registers together without
;; modifying the condition codes.
;;
;; So we do not expose the condition codes until after reload.  The
;; general approach is to have a define_insn_and_split for the basic
;; operation with no condition codes in the pattern (to give the
;; optimizers maximal freedom).  The splitter adds a clobber of the
;; condition codes after reload.  There is a distinct pattern which
;; sets the condition codes.
;;
;; As noted, data movement does not affect condition codes.
;;
;; Arithmetic generally set the codes in the expected ways, with mul
;; instructions being a notable outlier.  div instructions generally
;; do the right thing, except when the output registers are the same
;; when the flags do not get set.  We just assume they're clobbered
;; for div instructions to avoid MD bloat with marginal benefit
;;
;; The bit manipulation instructions (clr1, not1, set1) set condition
;; codes, but not in a useful way (they're set to the prior status of
;; the bit).  So we just model those as clobbers.  tst1 does set the
;; condition codes in a useful way.  We could perhaps do better with
;; these by noting they only modify the Z flag, it doesn't seem worth
;; the effort.
;;
;; Byte swaps seem to change the condition codes, but I haven't tried
;; to describe how.
;;
;; I have no documentation on the rotate instructions.  They likely
;; set the condition codes, but I've left them as clobbers for now.

;; movdi

(define_expand "movdi"
  [(set (match_operand:DI 0 "general_operand")
	(match_operand:DI 1 "general_operand"))]
  "TARGET_C33E3V5_UP"
  {
    /* One of the ops has to be in a register or 0.  */
    if (!register_operand (operand0, DImode)
        && !register_operand (operand1, DImode))
      operands[1] = copy_to_mode_reg (DImode, operand1);

    if (register_operand (operand0, DImode)
	&& (CONST_INT_P (operands[1]) || CONST_DOUBLE_P (operands[1])))
      {
        int i;

        for (i = 0; i < UNITS_PER_WORD * 2; i += UNITS_PER_WORD)
          emit_move_insn (simplify_gen_subreg (SImode, operands[0], DImode, i),
                          simplify_gen_subreg (SImode, operands[1], DImode, i));
        DONE;
      }
  }
)

(define_insn "*movdi_internal"
  [(set (match_operand:DI 0 "nonimmediate_operand" "=r,e!r,m")
	(match_operand:DI 1 "nonimmediate_operand"  "r,m,e!r"))]
  "TARGET_C33E3V5_UP
   || (register_operand (operands[0], DImode) && register_operand (operands[1], DImode))"
  { return c33_gen_movdi (operands); }
  [(set_attr "length" "4,12,12")
   (set_attr "type" "other,load,store")])

(define_expand "movsi"
  [(set (match_operand:SI 0 "general_operand" "")
	(match_operand:SI 1 "general_operand" ""))]
  ""
  {
    /* One side has to be a register.  There is no hardwired zero register on
       this target, so unlike the V850 a store of 0 is not a special case --
       the constant has to be loaded first.  */
    if (!register_operand (operands[0], SImode)
	&& !register_operand (operands[1], SImode))
      operands[1] = copy_to_mode_reg (SImode, operands[1]);

    /* No HIGH/LO_SUM splitting: xld.w takes a full 32-bit immediate and the
       assembler emits the two ext prefixes it needs.  The V850 had to build
       wide constants with movhi/movea because it had no such form.  */
  })

;; %sp is system register 1, so moving it uses the ld.w special-register
;; forms (RD,SS and SD,RS2).  Same mnemonic and same syntax as a general
;; register copy -- the assembler picks the encoding from the operand -- so
;; these are just two more alternatives rather than separate patterns.
;;
;; They have to exist.  When the frame or arg pointer is eliminated to %sp
;; and the result must reach an "r" operand, this is the only way to get it
;; there; without it LRA reloads the copy, the reload needs the same copy,
;; and it recurses until it hits the reload limit.

(define_insn "*movsi_internal"
  [(set (match_operand:SI 0 "nonimmediate_operand" "=r,r,r,m,r,f")
	(match_operand:SI 1 "movsi_source_operand"  " r,i,m,r,f,r"))]
  "register_operand (operands[0], SImode)
   || register_operand (operands[1], SImode)"
{
  return output_move_single (operands);
}
  [(set_attr "length" "2,6,6,6,2,2")
   (set_attr "type" "other,other,load,store,other,other")])

(define_insn "*c33_clr1_1"
  [(set (match_operand:QI 0 "memory_operand" "=m")
	(subreg:QI
	  (and:SI (subreg:SI (match_dup 0) 0)
		  (match_operand:QI 1 "not_power_of_two_operand" "")) 0))
   (clobber (reg:CC CC_REGNUM))]
  ""
{
  rtx xoperands[2];
  xoperands[0] = operands[0];
  xoperands[1] = GEN_INT (~INTVAL (operands[1]) & 0xff);
  output_asm_insn ("clr1 %M1,%0", xoperands);
  return "";
}
  [(set_attr "length" "4")
   (set_attr "type" "bit1")])

(define_insn "*c33_clr1_2"
  [(set (match_operand:HI 0 "indirect_operand" "=m")
	(subreg:HI
	  (and:SI (subreg:SI (match_dup 0) 0)
		  (match_operand:HI 1 "not_power_of_two_operand" "")) 0))
   (clobber (reg:CC CC_REGNUM))]
  ""
{
  int log2 = exact_log2 (~INTVAL (operands[1]) & 0xffff);

;; Instructions using cc0 aren't allowed to have input reloads, so we must
;; hide the fact that this instruction uses cc0.  We do so by including the
;; compare instruction inside it.

(define_expand "movsicc"
  [(set (match_operand:SI 0 "register_operand" "=r")
	(if_then_else:SI
	 (match_operand 1 "comparison_operator")
	 (match_operand:SI 2 "reg_or_const_operand" "rJ")
	 (match_operand:SI 3 "reg_or_const_operand" "rI")))]
  "(TARGET_C33E_UP)"
  {
    /* Make sure that we have an integer comparison...  */
    if (GET_MODE (XEXP (operands[1], 0)) != CCmode
        && GET_MODE (XEXP (operands[1], 0)) != SImode)
      FAIL;

    if ((GET_CODE (operands[2]) == CONST_INT
	&& GET_CODE (operands[3]) == CONST_INT))
      {
	int o2 = INTVAL (operands[2]);
	int o3 = INTVAL (operands[3]);

	if (o2 == 1 && o3 == 0)
	  FAIL;   /* setf */
	if (o3 == 1 && o2 == 0)
	  FAIL;   /* setf */
	if (o2 == 0 && (o3 < -16 || o3 > 15) && exact_log2 (o3) >= 0)
	  FAIL;   /* setf + shift */
	if (o3 == 0 && (o2 < -16 || o2 > 15) && exact_log2 (o2) >=0)
	  FAIL;   /* setf + shift */
	if (o2 != 0)
	  operands[2] = copy_to_mode_reg (SImode, operands[2]);
	if (o3 !=0 )
	  operands[3] = copy_to_mode_reg (SImode, operands[3]);
      }
    else
      {
	if (GET_CODE (operands[2]) != REG)
	  operands[2] = copy_to_mode_reg (SImode,operands[2]);
	if (GET_CODE (operands[3]) != REG)
	  operands[3] = copy_to_mode_reg (SImode, operands[3]);
      }
  })

   (match_operand 3 "" "") (match_operand 4 "" "")]
  ""
  {
    rtx reg = gen_reg_rtx (SImode);
    rtx tableaddress = gen_reg_rtx (SImode);
    rtx test;
    rtx mem;

    /* Subtract the lower bound from the index.  */
    emit_insn (gen_subsi3 (reg, operands[0], operands[1]));

    /* Compare the result against the number of table entries;
       branch to the default label if out of range of the table.  */
    test = gen_rtx_fmt_ee (GTU, VOIDmode, reg, operands[2]);
    emit_jump_insn (gen_cbranchsi4 (test, reg, operands[2], operands[4]));

    emit_insn (gen_ashlsi3 (reg, reg, GEN_INT (TARGET_BIG_SWITCH ? 2 : 1)));

    emit_insn (gen_movsi (tableaddress,

	rtx reg2 = gen_reg_rtx (HImode);
	emit_insn (gen_movhi (reg2, mem));
	emit_insn (gen_extendhisi2 (reg, reg2));
      }
    else
      emit_insn (gen_movsi (reg, mem));
    /* Add the table address.  */
    emit_insn (gen_addsi3 (reg, reg, tableaddress));
    /* Branch to the switch label.  */
    emit_jump_insn (gen_tablejump (reg, operands[3]));
    DONE;
  })

;; The C33 has no link register: call pushes the return address onto the
;; stack and ret pops it (core manual 2.4).  So there is nothing to clobber
;; here beyond the flags -- V850's (clobber (reg:SI 31)) was its link
;; register, and with FIRST_PSEUDO_REGISTER at 22 it named a *pseudo*, which
;; postreload rejects outright ("no spurious CLOBBERs of pseudo registers").
;;
;; Three forms of the direct call, differing only in reach:
;;   call  disp8   +/-  256 bytes, 2 bytes
;;   scall disp22  +/-   4 MB,     4 bytes  (assembler expands with one ext)
;;   xcall disp32  whole space,    6 bytes  (two ext prefixes)
;; and the indirect call  call %rb, 2 bytes.
;;
;; -mlong-calls selects xcall, which is what samo-lib builds with.  We never
;; emit the bare disp8 call: the assembler cannot relax, so picking it would
;; mean proving the callee is within 256 bytes.

(define_insn "call_internal_short"
  [(call (mem:QI (match_operand:SI 0 "call_address_operand" "S,r"))
	 (match_operand:SI 1 "general_operand" "g,g"))
   (clobber (reg:CC CC_REGNUM))]
  "! TARGET_LONG_CALLS"
  "@
   scall\t%0
   call\t%0"
  [(set_attr "length" "4,2")])

(define_insn "call_internal_long"

;; Stack pointer arithmetic.
;;
;; %sp is not a general register, and the C33 has dedicated forms for moving
;; it: add/sub %sp,imm10 scale the immediate by 4 and cannot be extended with
;; ext, so one instruction reaches +/-4092 bytes (core manual p64).  Anything
;; larger has to go through a general register and ld.w %sp,%rs.

(define_insn "add_sp_imm"
  [(set (reg:SI SP_REGNUM)
	(plus:SI (reg:SI SP_REGNUM)
		 (match_operand:SI 0 "const_int_operand" "i")))]
  "IN_RANGE (INTVAL (operands[0]), -4092, 4092)
   && (INTVAL (operands[0]) & 3) == 0"
{
  HOST_WIDE_INT n = INTVAL (operands[0]);
  operands[0] = GEN_INT ((n < 0 ? -n : n) / 4);
  return n < 0 ? "sub\t%%sp,%0" : "add\t%%sp,%0";
}
  [(set_attr "length" "2")])

;; Frames deeper than 4092 bytes.  add/sub %sp,imm10 cannot be extended, so
;; the adjustment has to be done in a general register.  xadd/xsub %rd,imm32
;; take any 32-bit constant (the assembler supplies the ext prefixes), so one
;; scratch is enough -- there is no need to materialise the delta separately.
;;
;; %r14 is call-clobbered and holds neither an incoming argument nor a return
;; value, so it is free in both the prologue and the epilogue.

(define_insn "add_sp_big"
  [(set (reg:SI SP_REGNUM)
	(plus:SI (reg:SI SP_REGNUM)
		 (match_operand:SI 0 "const_int_operand" "i")))
   (clobber (reg:SI 14))]
  ""
{
  HOST_WIDE_INT n = INTVAL (operands[0]);
  operands[0] = GEN_INT (n < 0 ? -n : n);
  return n < 0
	 ? "ld.w\t%%r14,%%sp\n\txsub\t%%r14,%0\n\tld.w\t%%sp,%%r14"
	 : "ld.w\t%%r14,%%sp\n\txadd\t%%r14,%0\n\tld.w\t%%sp,%%r14";
}
  [(set_attr "length" "10")])

;; Block save/restore of the callee-saved registers.
;;
;; pushn %rN pushes the whole block %r0..%rN, decrementing %sp by 4 for each
;; (core manual 2.4.2); popn %rN is the inverse (2.4.3).  Operand 0 is N.
;; The effect on %sp and memory is described by the unspec_volatile plus the
;; explicit clobber, which is enough for the frame-related notes the
;; prologue attaches.

(define_insn "pushn"
  [(unspec_volatile [(match_operand 0 "const_int_operand" "i")]
		    UNSPECV_PUSHN)
   (clobber (reg:SI SP_REGNUM))]
  ""
  "pushn\t%%r%0"
  [(set_attr "length" "2")])

(define_insn "popn"
  [(unspec_volatile [(match_operand 0 "const_int_operand" "i")]
		    UNSPECV_POPN)
   (clobber (reg:SI SP_REGNUM))]
  ""
  "popn\t%%r%0"
  [(set_attr "length" "2")])

;; An interrupt pushes both PC and PSR, and reti pops both (core manual
;; 2.4.4).  There is nothing else to do: compute_register_save_size already
;; widens the saved block to every register the handler touches, and the
;; prologue saves it with the same pushn/popn a normal function uses.
;;
;; Everything V850 had here -- callt_save_interrupt, save_all_interrupt and
;; friends -- was for its ep/gp/callt model and its 32 registers, and named
;; registers that do not exist on this target (r30 was ep; with
;; FIRST_PSEUDO_REGISTER at 22 those are pseudo numbers, which postreload
;; rejects).  None of it was reachable: nothing in c33.cc ever generated it.

(define_insn "return_interrupt"
  [(return)
   (unspec_volatile [(const_int 0)] UNSPECV_RETI)]
  ""
  "reti"
  [(set_attr "length" "2")])
